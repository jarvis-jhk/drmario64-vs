cmake_minimum_required(VERSION 3.16)
project(drmario64-vs VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Off for source and development builds: only published release binaries post anonymous faults.
option(DM_FAULT_REPORTS "Let the game post anonymous fault reports" OFF)
option(DM_BUILD_APP "Build the SDL2 frontend (the core and its tests need no SDL)" ON)

if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

# Release builds pass the tag's version; everything else says which source version it is.
set(DM_VERSION "${PROJECT_VERSION}-dev" CACHE STRING "Version shown in the game")
configure_file(src/version.h.in ${CMAKE_BINARY_DIR}/generated/version.h @ONLY)

# Deterministic engine core: no SDL, no clock, no I/O. Everything the original game
# decides happens here, so it can be golden-tested headless frame by frame.
add_library(dmcore STATIC src/core/rng.c src/core/setup.c src/core/vs.c)
target_include_directories(dmcore PUBLIC src/core ${CMAKE_BINARY_DIR}/generated)

enable_testing()
add_executable(test_core tests/test_core.c)
target_link_libraries(test_core PRIVATE dmcore)
add_test(NAME core COMMAND test_core ${CMAKE_SOURCE_DIR}/tests/fixtures)
# Golden traces: the VS engine against the original's per-tick state (tools/oracle).
add_executable(test_vs tests/test_vs.c)
target_link_libraries(test_vs PRIVATE dmcore)
add_test(NAME vs_traces COMMAND test_vs ${CMAKE_SOURCE_DIR}/tests/movies)
set_tests_properties(vs_traces PROPERTIES TIMEOUT 900)
# The frontend logic (menus, setup, replay) is SDL-free too, so a session is a test.
add_library(dmfront STATIC src/app/front.c src/app/bot.c src/app/lockstep.c)
target_include_directories(dmfront PUBLIC src/app)
target_link_libraries(dmfront PUBLIC dmcore)
add_executable(test_front tests/test_front.c)
target_link_libraries(test_front PRIVATE dmfront)
add_test(NAME front_session COMMAND test_front)
# Online lockstep between two simulated machines over a bad link.
add_executable(test_lockstep tests/test_lockstep.c)
target_link_libraries(test_lockstep PRIVATE dmfront)
add_test(NAME lockstep COMMAND test_lockstep)
# Sound effects from the player's ROM: skipped unless DM_ROM names one.
add_executable(test_romsfx tests/test_romsfx.c src/app/romsfx.c)
target_include_directories(test_romsfx PRIVATE src/app)
add_executable(sfxdump tools/sfxdump.c src/app/romsfx.c)
target_include_directories(sfxdump PRIVATE src/app)
if(NOT MSVC)
  target_link_libraries(test_romsfx PRIVATE m)
  target_link_libraries(sfxdump PRIVATE m)
endif()
add_test(NAME romsfx COMMAND test_romsfx)

if(DM_BUILD_APP)
  find_package(SDL2 REQUIRED)
  if(DM_FAULT_REPORTS)
    set(DM_FAULT_REPORTS_VALUE 1)
  else()
    set(DM_FAULT_REPORTS_VALUE 0)
  endif()
  add_executable(drmario64-vs WIN32 src/app/main.c src/app/fault.c src/app/draw.c src/app/render.c src/app/sfx.c src/app/romsfx.c
                 src/app/online.c src/app/netui.c)
  target_compile_definitions(drmario64-vs PRIVATE DM_FAULT_REPORTS=${DM_FAULT_REPORTS_VALUE})
  target_link_libraries(drmario64-vs PRIVATE dmfront)
  # Online versus through Warpzone: a pinned libwarpzone release, checked against its
  # SHA-256 like any other download. -DDM_WARPZONE=OFF builds without it.
  option(DM_WARPZONE "Online versus and the ranked ladder through Warpzone" ON)
  set(DM_WARPZONE_DIR "" CACHE PATH "An unpacked libwarpzone package to use instead of downloading one")
  if(DM_WARPZONE)
    set(WZ_VERSION 0.1.1)
    if(WIN32)
      set(WZ_PACKAGE libwarpzone-${WZ_VERSION}-x86_64-windows-gnu)
      set(WZ_FILE ${WZ_PACKAGE}.zip)
      set(WZ_SHA256 c3ebac9eb6a77dbfab5c171013908f5a8ca47a4d6488a6fb11601ab4e15a638a)
    else()
      set(WZ_PACKAGE libwarpzone-${WZ_VERSION}-x86_64-linux-gnu)
      set(WZ_FILE ${WZ_PACKAGE}.tar.gz)
      set(WZ_SHA256 dd758e80a69b6b7ed755619a570b0f860412bc3ab69eda35d05b93b9084ba96b)
    endif()
    if(NOT DM_WARPZONE_DIR)
      set(WZ_DEPS ${CMAKE_BINARY_DIR}/_deps)
      if(NOT EXISTS ${WZ_DEPS}/${WZ_PACKAGE}/include/warpzone.h)
        file(DOWNLOAD https://github.com/jarvis-jhk/warpzone/releases/download/v${WZ_VERSION}/${WZ_FILE} ${WZ_DEPS}/${WZ_FILE}
             EXPECTED_HASH SHA256=${WZ_SHA256} TLS_VERIFY ON)
        execute_process(COMMAND ${CMAKE_COMMAND} -E tar xf ${WZ_DEPS}/${WZ_FILE} WORKING_DIRECTORY ${WZ_DEPS} RESULT_VARIABLE WZ_UNPACKED)
        if(NOT WZ_UNPACKED EQUAL 0)
          message(FATAL_ERROR "Cannot unpack ${WZ_FILE}")
        endif()
      endif()
      set(DM_WARPZONE_DIR ${WZ_DEPS}/${WZ_PACKAGE})
    endif()
    target_compile_definitions(drmario64-vs PRIVATE DM_WARPZONE)
    target_include_directories(drmario64-vs PRIVATE ${DM_WARPZONE_DIR}/include)
    target_link_libraries(drmario64-vs PRIVATE ${DM_WARPZONE_DIR}/lib/libwarpzone.a)
    if(WIN32)
      target_link_libraries(drmario64-vs PRIVATE kernel32 advapi32 ntdll userenv ws2_32 bcrypt dbghelp)
    else()
      target_link_libraries(drmario64-vs PRIVATE pthread dl m)
    endif()
  endif()
  if(NOT MSVC)
    target_link_libraries(drmario64-vs PRIVATE m)
  endif()
  if(TARGET SDL2::SDL2main)
    target_link_libraries(drmario64-vs PRIVATE SDL2::SDL2main)
  endif()
  if(TARGET SDL2::SDL2)
    target_link_libraries(drmario64-vs PRIVATE SDL2::SDL2)
  else()
    target_include_directories(drmario64-vs PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_libraries(drmario64-vs PRIVATE ${SDL2_LIBRARIES})
  endif()
  install(TARGETS drmario64-vs RUNTIME DESTINATION bin)
  add_test(NAME app_smoke COMMAND drmario64-vs --headless --frames 120)
  set_tests_properties(app_smoke PROPERTIES TIMEOUT 30 ENVIRONMENT "DM_NO_REPORT=1;SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
  # Two bots play a round through the real frontend and the renderer draws it (software).
  add_test(NAME app_render COMMAND drmario64-vs --shot ${CMAKE_BINARY_DIR}/app_render.bmp --demo --until round --frames 40000 --size 960x720)
  # The Steam artwork the release packages.
  add_test(NAME app_art COMMAND drmario64-vs --art ${CMAKE_BINARY_DIR})
  set_tests_properties(app_art PROPERTIES TIMEOUT 60 ENVIRONMENT "DM_NO_REPORT=1;SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
  set_tests_properties(app_render PROPERTIES TIMEOUT 120 ENVIRONMENT "DM_NO_REPORT=1;SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
  # The online screens (menu, code entry, open room, a ranked game's banner), drawn from made-up state.
  foreach(screen online-menu online-code online-room online-game)
    add_test(NAME app_${screen} COMMAND drmario64-vs --shot ${CMAKE_BINARY_DIR}/app_${screen}.bmp --demo --until ${screen} --frames 600 --size 960x720)
    set_tests_properties(app_${screen} PROPERTIES TIMEOUT 60 ENVIRONMENT "DM_NO_REPORT=1;SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
  endforeach()
endif()
